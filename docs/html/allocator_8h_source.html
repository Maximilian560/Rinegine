25 // std::cout << "push" << std::endl;
26 RG_LOG_LOCK_MEM("Push " + std::to_string(in) + " page(s)");
27 Kernel::Allocator::_map *addr = s_map(in);
28
29 if (addr != nullptr) {
30 Kernel::Allocator::_map *mmin = rg_min(addr, _main_map);
31 Kernel::Allocator::_map *mmax = rg_max(addr, _main_map);
32 if (size_t(mmax) - size_t(mmin) == Rinegine::Kernel::Lock::page_size * in) {
33 // std::cout << "prev: " << (int)mmax->magnum[2] << std::endl;
34 mmin->_mem.next = &mmax->_mem;
35 mmin->magnum[2] += mmax->magnum[2];

37 mmax->magnum[1] = 0;
38 mmax->magnum[2] = 0;
39
40 _main_map = mmin;
41 if (_main_map->magnum[2] == UINT8_MAX) {
42 RG_LOG_LOCK_CRITICAL("Strange, but the size of the page pool has "
43 "become too large, the number of pages is " +
44 std::to_string(size_t(_main_map->magnum[2])));
45 }
46 } else {
47 RG_LOG_LOCK_MEM("Error! Failed to add page to shared heap!");
48 RG_LOG_LOCK_DEBUG("size_t(mmax) - size_t(mmin) = " +

51 RG_LOG_LOCK_CRITICAL("Mem error, see previous log");
